/**** sample code for triggering useRef using useState
 * generally avoid using useState as it rerenders the entire page frequently on state change
 * if necessary to use useState, please trigger manually.
 ****/

import { useRef, useState } from "react";

const MyComponent = () => {
  const bufferRef = useRef(new ArrayBuffer(1024 * 1024)); // Large buffer
  const [, forceRender] = useState(0); // Dummy state

  const updateBuffer = () => {
    bufferRef.current = new ArrayBuffer(1024 * 1024); // Update ref
    forceRender((prev) => prev + 1); // Force re-render
  };

  return (
    <div>
      <p>Buffer updated!</p>
      <button onClick={updateBuffer}>Update Buffer</button>
    </div>
  );
};
